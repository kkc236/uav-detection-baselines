# BQP-DETR G0 Feasibility Audit Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build and run a read-only G0 audit that determines whether missed tiny-object stock candidates in RT-DETR-L ranks 301-420 can safely replace clear background candidates in ranks 251-300.

**Architecture:** Keep the scientific logic in a pure PyTorch module and isolate Ultralytics integration in a capture adapter plus a validator-based CLI. The adapter calls the unmodified stock decoder path first, recomputes only the extra ranked candidates under `no_grad`, and rejects any mismatch with the stock Top-300 outputs. The CLI uses a hashed training-subset YAML, carries VisDrone ignore boxes through the same geometric transforms, and writes a manifest, per-image JSONL, and a frozen pass/fail summary.

**Tech Stack:** Python 3.10, PyTorch 2.5.1+cu121, Ultralytics 8.4.90, pytest, RTX 4090, JSON/JSONL.

---

- Create `src/bqp_g0.py`: tensor contracts, NWD/IoA geometry, per-image pair mining, aggregation, and gate evaluation.

```python
from __future__ import annotations

from dataclasses import dataclass

import torch


@dataclass(frozen=True)
class BQPG0Config:
    selected_k: int = 300
    insider_start: int = 250
    outsider_stop: int = 420
    tiny_area_threshold: float = (32.0 / 640.0) ** 2
    nwd_distance_constant: float = 12.8 / 640.0
    cover_threshold: float = 0.50
    background_threshold: float = 0.10
    ignore_ioa_threshold: float = 0.50
    minimum_missed: int = 200
    minimum_pairs: int = 60
    minimum_promotable_ratio: float = 0.30


def tiny_mask(boxes_xywh: torch.Tensor, config: BQPG0Config) -> torch.Tensor:
    return boxes_xywh[:, 2:].prod(dim=-1) < config.tiny_area_threshold


def normalized_wasserstein_similarity(
    boxes1_xywh: torch.Tensor,
    boxes2_xywh: torch.Tensor,
    *,
    distance_constant: float,
) -> torch.Tensor:
    delta_center = boxes1_xywh[:, None, :2] - boxes2_xywh[None, :, :2]
    delta_size = (boxes1_xywh[:, None, 2:] - boxes2_xywh[None, :, 2:]) * 0.5
    distance = torch.sqrt(
        delta_center.square().sum(-1) + delta_size.square().sum(-1)
    )
    return torch.exp(-distance / distance_constant)


def intersection_over_area_xywh(
    candidates_xywh: torch.Tensor, regions_xywh: torch.Tensor
) -> torch.Tensor:
    if regions_xywh.numel() == 0:
        return candidates_xywh.new_zeros((len(candidates_xywh), 0))
    candidate_xyxy = torch.cat(
        (
            candidates_xywh[:, :2] - candidates_xywh[:, 2:] / 2,
            candidates_xywh[:, :2] + candidates_xywh[:, 2:] / 2,
        ),
        dim=-1,
    )
    region_xyxy = torch.cat(
        (
            regions_xywh[:, :2] - regions_xywh[:, 2:] / 2,
            regions_xywh[:, :2] + regions_xywh[:, 2:] / 2,
        ),
        dim=-1,
    )
    left_top = torch.maximum(candidate_xyxy[:, None, :2], region_xyxy[None, :, :2])
    right_bottom = torch.minimum(candidate_xyxy[:, None, 2:], region_xyxy[None, :, 2:])
    intersection = (right_bottom - left_top).clamp_min(0).prod(-1)
    candidate_area = candidates_xywh[:, 2:].prod(-1).clamp_min(1e-12)
    return intersection / candidate_area[:, None]
```

Add:

```python
def evaluate_gate(
    *,
    missed: int,
    pairs: int,
    oracle_net: int,
    top300_exact: bool,
    all_finite: bool,
    ignored_negative_count: int,
    config: BQPG0Config,
) -> dict[str, object]:
    ratio = pairs / missed if missed else 0.0
    checks = {
        "top300_exact": top300_exact,
        "all_finite": all_finite,
        "minimum_missed": missed >= config.minimum_missed,
        "minimum_pairs": pairs >= config.minimum_pairs,
        "minimum_promotable_ratio": ratio >= config.minimum_promotable_ratio,
        "positive_oracle_net": oracle_net > 0,
        "no_ignored_negatives": ignored_negative_count == 0,
    }
    failed = [name for name, passed in checks.items() if not passed]
    return {
        "passed": not failed,
        "status": "G0_PASS" if not failed else "G0_FAIL",
        "checks": checks,
        "failed_reasons": failed,
    }
```

The oracle result must be computed from unique per-image swaps, not by assuming
