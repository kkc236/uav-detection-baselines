# LPR-G v2 30-Epoch Cutoff Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Preserve the 50-epoch learning-rate and mosaic schedule while ending each seed0 10% screening arm immediately after epoch 30 has been published and remotely verified, then compare the exact epoch 1-30 evidence before optionally launching fresh 100-epoch full-data arms.

**Architecture:** Keep `trainer.args.epochs == 50` for screening and add an operational cutoff callback after diagnostics, common-state audit, and GitHub publication. The supervisor and evidence readers treat 30 verified epochs as the screen completion boundary, while formal experiments remain 100 epochs. Migrate the already-running control arm by stopping its exact process group only after its verified epoch-30 ledger exists, preserve all original evidence, and let the updated supervisor reuse that arm.

**Tech Stack:** Python 3.10, Ultralytics 8.4.90 callbacks, PyTorch 2.5.1, pytest, JSONL publication ledgers, Paramiko SSH, Git/GitHub Releases.

---

```python
SCREEN_SCHEDULE_EPOCHS = 50
SCREEN_CUTOFF_EPOCHS = 30
FORMAL_EPOCHS = 100

def _scientific_epochs(stage: str) -> int:
    return SCREEN_SCHEDULE_EPOCHS if stage == "screen" else FORMAL_EPOCHS

def cutoff_after_verified_publication(trainer, *, args: argparse.Namespace) -> bool:
    if args.preflight or args.stage != "screen":
        return False
    completed_epoch = int(trainer.epoch + 1)
    if completed_epoch < SCREEN_CUTOFF_EPOCHS:
        return False
    record = getattr(trainer, "lpr_g_publication_record", None)
    valid = (
        isinstance(record, dict)
        and int(record.get("completed_epoch", -1)) == SCREEN_CUTOFF_EPOCHS
        and record.get("verified") is True
    )
    if completed_epoch != SCREEN_CUTOFF_EPOCHS or not valid:
        raise RuntimeError("screen cutoff requires verified epoch 30 publication")
    trainer.stop = True
    return True
```

```python
def _write_arm(path: Path, *, method: bool, epochs: int = 30) -> None:
    path.mkdir(parents=True)
    with (path / "results.csv").open("w", newline="", encoding="utf-8") as stream:
        writer = csv.DictWriter(
            stream,
            fieldnames=("epoch", "metrics/mAP50-95(B)", "metrics/mAP50(B)"),
        )
        writer.writeheader()
        for epoch in range(1, epochs + 1):
            writer.writerow({
                "epoch": epoch,
                "metrics/mAP50-95(B)": epoch / 1000 + (0.001 if method else 0.0),
                "metrics/mAP50(B)": epoch / 500 + (0.001 if method else 0.0),
            })
    diagnostics = [{
        "epoch": epoch,
        "map75": epoch / 2000 + (0.001 if method else 0.0),
        "loss_bbox_refine": 0.3 if method else None,
        "loss_giou_refine": 0.4 if method else None,
        "gate_p95": 0.01 if method else None,
        "residual_rms": 0.001 if method else None,
        "lpr_g_gradient_norm": 1.0 if method else None,
    } for epoch in range(1, epochs + 1)]
    (path / "lpr_g_diagnostics.jsonl").write_text(
        "".join(json.dumps(row) + "\n" for row in diagnostics), encoding="utf-8"
    )
    audits = [{
        "epoch": epoch,
        "common_model_sha256": f"{epoch:064x}",
        "common_optimizer_sha256": f"{epoch + 100:064x}",
    } for epoch in range(1, epochs + 1)]
    (path / "common_state_audit.jsonl").write_text(
        "".join(json.dumps(row) + "\n" for row in audits), encoding="utf-8"
    )
```
